로그인 코드

  <!DOCTYPE html>
  <html lang="ko">
  <head>
      <title>로그인 창</title>
  </head>
  <body>
      <h2 style="color: rgb(26, 191, 80);">NAVER</h2>
  
      <input type="text" id="아이디" placeholder="Username">
      <br>
      <input type="password" id="비밀번호" placeholder="password">
      <br>
      <input type="button" value="Sign in" onclick="로그인()">
  
      <script>
          const 아이디 = "홍길동";
          const 비번 = 11111;
  
          function 로그인() {
              let 입력한아이디 = document.getElementById("아이디").value;
              let 입력한비번 = document.getElementById("비밀번호").value;
  
              // console.log(`${입력한아이디} ${입력한비번}`);
  
              if (아이디 == 입력한아이디) {
                  if (비번 == 입력한비번) {
                      // location.href = "http://naver.com";
                      open("http://naver.com", "길동", "width = 350, height = 400, left = 50, top = 10, scrollbars = no");
                  } else {
                      alert("비번 고쳐.");
                      location.reload();
                  }
              } else {
                  alert("아이디 고쳐.");
                  location.reload();
              }
              
          }
  
      </script>
  
  </body>
  </html>


더하기 코드 (안됌)

  <!DOCTYPE html>
  <html lang="en">
  <head>
      <title>Document</title>
  </head>
  <body>
      <input type="number" id="숫자1">
      <input type="number" id="숫자2">
      <input type="number" id="결과">
      <input type="button" value= "더하기" onclick="더하기()">
  
      <script>
           
          /* 
          function 구구단(){}
      //  아래부터 호이스팅(끌어올린다)이 됌
          let 구구단2 = function(){}
          let 구구단3 = () => {} 
          */
  
          let 더하기 = () => {
              let num1 = parent(document.getElementById("숫자1").value);
              let num2 = parent(document.getElementById("숫자2").value);
  
              document.getElementById("결과").value = num1 + num2;
          }
  
      </script>
