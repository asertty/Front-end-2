                    <input type="checkbox" value="다크모드" id="모드변경">
                </div>
            </header>
        
            <nav>
                <div class="logo">
                    <a href="#">
                        <img src="logo.png" alt="로고">
                    </a>
                </div>
                <!-- ul>(li>a[href="#"])*3 -->

                <ul class="menu">

            <section>
                <h2><a href="#">Home</a></h2>
                
                <p>
                    Lorem ipsum dolor sit, amet consectetur adipisicing elit.<br>
                    Incidunt quibusdam iusto, alias est perferendis aut ipsa itaque<br>

                </p>

        <script>
            let 체크상자 = document.getElementById("모드변경");
    
            체크상자.addEventListener("change", function(){
    
                if(체크상자.checked == true){
                    // 체크되어있음
                    document.getElementsByTagName('nav')[0].style.backgroundColor = "black";
                    document.body.style.backgroundColor = "black";
    
                    document.getElementsByTagName('section')[0].style.backgroundColor = 'black';
                    document.getElementsByTagName('h2')[0].style.Color = 'black';
    
                    document.getElementsByTagName('p')[0].style.backgroundColor = '#372D2B';
                    document.getElementsByTagName('p')[0].style.Color = '#fae5ac';
    
                    let 리스트 = document.getElementsByTagName('li');
                    for(let i = 0; i <= 리스트.length; i++){
                        리스트[i].style.backgroundColor = "#555555";
                        리스트[i].children[0].backgroundColor = "gold";
                        
                    }
                } else {
                    // 체크안되어있음
                    document.getElementsByTagName('nav')[0].style.backgroundColor = "white";
                    document.body.style.backgroundColor = "white";
    
                    document.getElementsByTagName('section')[0].style.backgroundColor = 'white';
                    document.getElementsByTagName('h2')[0].style.Color = 'black';
                    document.getElementsByTagName('h2')[0].children[0].style.Color = 'black';
    
                    document.getElementsByTagName('p')[0].style.backgroundColor = 'beige';
                    document.getElementsByTagName('p')[0].style.Color = 'black';
    
                    let 리스트 = document.getElementsByTagName('li');
                    for(let i = 0; i <= 리스트.length; i++){
                        리스트[i].style.backgroundColor = "#dbddf7";
                        리스트[i].children[0].backgroundColor = "#black";
                        
                    }
        
                }
    
            })
    
    
        </script>
    </body>
    </html>
